# Rook 3K configuration
#
# Read by src/rook3k.py and src/make_3mf.py. Lines are key=value, and # starts a comment.
# All dimensions are in millimetres. A key that is left out or left empty keeps its default.
# The values below are the defaults.

# --- Mainboard -----------------------------------------------------------------
# denise_itx  DeniseITX, with its I/O cut-outs built into the rear panel and two Zorro slots.
# pc_itx      Any Mini-ITX board: a standard ATX aperture for the loose I/O shield. No Zorro slots. Not test printed.
# alicia_itx  Alicia 1200, with its I/O cut-outs built into the rear panel. No Zorro slots.
#             The dimensions have not been verified against an actual Alicia yet.
mainboard=denise_itx

# --- Case size ---------------------------------------------------------------
# Outer height (without the foot), depth and width of the case.
# Minimum sizes apply and depend on the options below. The script stops with an error if a value
# is too low, and warns if the case exceeds the Bambu Lab H2D single nozzle volume (325 x 320 x 325).
# The defaults are the smallest practical case with a legacy floppy drive.
# The minimum depth is 210. It gives a warning, because the default of 211 leaves 1 mm more
# between the pins of the power and reset switches and the mainboard.
rook_height=227
rook_depth=211
rook_width=117

# Corner radius seen from the front, 0.5 to 6.
rook_corner_radius=4

# --- Bottom -------------------------------------------------------------------
# foot_grill  One foot under the whole case, built as a stack of horizontal grills.
# foot_pads   Four separate round feet, each with a recess for a TPU pad. The pads are generated as well.
# none        Nothing under the case, and no screw holes for a foot.
bottom=foot_grill

# Grill foot. Its height is grills x height + (grills - 1) x gap.
rook_foot_grill_height=3
rook_foot_grill_gap=2.5
rook_foot_grills=3

# Feet with pads. Diameter 12 to 20.
bottom_pad_height=10
bottom_pad_diameter=18

# --- Front panel ----------------------------------------------------------------
# The recessed chamfer field with the flat groove. false gives a flat front.
chamfer_recess_enable=true

# The thin decorative groove on the left chamfer. Needs the chamfer recess.
deco_groove_enable=true

# Recess for a 25 x 25 mm case badge.
pc_badge_recess_enable=true

# Power and reset buttons. Front buttons sit in a pocket in the front panel and press MX keyboard switches
# behind it. Each button is placed on its own.
# button_power:
#   front       Power button in the front.
#   front_wide  One tall power button that fills the front pocket. Needs button_reset=back or none.
#   back        Planned, not available yet.
#   back_wide   Planned, not available yet. Needs button_reset=front or none.
#   none        No power button. Gives a warning, the DeniseITX needs a pulse on its power input to turn on.
# button_reset:
#   front       Reset button in the front.
#   back        Planned, not available yet.
#   none        No reset button.
# With no button in the front there is no pocket and no switch plate, and the case can be 20 mm shallower.
button_power=front
button_reset=front

# The two LED windows, power at the top and drive below it. The defaults fit 2 x 5 mm rectangular LEDs.
led_power_enable=true
led_drive_enable=true
led_window_width=2.2
led_window_length=5

# --- Floppy drive ---------------------------------------------------------------
# Legacy 3.5" floppy drive. false removes the disk slot and the eject cut-out in the front,
# the shelf on the side panel and the test gauge.
floppy_legacy_enable=true

# Height of the mounting holes above the bottom plate of the drive.
# 5.0 is verified for a Chinon FZ-354 without face plate. Print the test gauge to check other drives.
floppy_mount_hole_height=5

# --- Rear panel -----------------------------------------------------------------
# The two Zorro slots, slot 1 at the bottom. Only for mainboard=denise_itx.
# open    Opening with a lip and a blanking cover, for any Zorro card.
# closed  Solid panel.
# custom  Solid panel with your own cut-outs. The script cuts custom1.step or custom2.step, found next to
#         this file, out of the slot. Draw the cut-outs on out/zorro_slot_template.step, which shows the
#         60 x 14 mm field and its origin. An optional custom1_add.step or custom2_add.step is added afterwards.
zorro_slot_1=open
zorro_slot_2=open

# Cut-out for a DE-15 panel connector, for example from an Indivision scan doubler.
back_port_vga_enable=false

# Material left in the recess behind the VGA cut-out, 0.8 to 1.2. With the cut-out enabled, a test gauge
# with 1.2, 1.0 and 0.8 is written to out/vga_test_gauge.step. All three fitted the connector it was tried with.
back_port_vga_rim_thickness=0.8

# Hole for the DC panel jack, and its diameter.
back_port_dc_enable=true
back_port_dc_diameter=8.1

# --- Storage ------------------------------------------------------------------
# Four ears on the inside of the left side panel for a 2.5" SSD, above the floppy shelf. The drive is
# screwed through the ears into its side holes, so no screws show on the outside. Not test printed.
ssd_mount_enable=false

# --- Heat-set inserts -----------------------------------------------------------
# The defaults are for CNC Kitchen M3 x 3.0 inserts. The depth is the depth of the hole,
# which leaves room behind the insert for displaced plastic.
heatinsert_diameter=4
heatinsert_depth=4.5
heatinsert_wall_thickness=1.6
